Table Male {
  id integer [primary key, ref: - Person.id]
}

Table Female {
  id integer [primary key, ref: - Person.id]
}

Table Person {
  id integer [primary key] 
  first_name varchar
  last_name varchar
  mother_id integer [ref: - Female.id]
  father_id integer [ref: - Male.id]
}

Table Marriage {
  husband_id integer [unique, ref: - Male.id]
  wife_id integer [unique, ref: - Female.id]
}
